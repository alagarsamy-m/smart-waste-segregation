import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense
from tensorflow.keras.preprocessing.image import ImageDataGenerator

# Image preprocessing
datagen = ImageDataGenerator(rescale=1./255, validation_split=0.2)

train_data = datagen.flow_from_directory('waste_dataset', target_size=(128, 128),
                                         batch_size=32, class_mode='categorical',
                                         subset='training')
val_data = datagen.flow_from_directory('waste_dataset', target_size=(128, 128),
                                       batch_size=32, class_mode='categorical',
                                       subset='validation')

# CNN model
model = Sequential([
    Conv2D(32, (3,3), activation='relu', input_shape=(128,128,3)),
    MaxPooling2D(2,2),
    Conv2D(64, (3,3), activation='relu'),
    MaxPooling2D(2,2),
    Flatten(),
    Dense(128, activation='relu'),
    Dense(4, activation='softmax')  # 4 classes: plastic, metal, glass, paper
])

model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])
model.fit(train_data, epochs=10, validation_data=val_data)




//sample demostration use

model.save('waste_classifier.h5')


import tensorflow as tf
import numpy as np
from tensorflow.keras.preprocessing import image
import matplotlib.pyplot as plt
import os

# Load the trained model
model = tf.keras.models.load_model('waste_classifier.h5')

# Map class indices manually (match your training folder structure)
class_names = ['glass', 'metal', 'paper', 'plastic']  # Adjust if needed

# Load and preprocess a single image
def load_and_prepare_image(img_path):
    img = image.load_img(img_path, target_size=(128, 128))
    img_array = image.img_to_array(img) / 255.0  # Rescale to [0,1]
    img_array = np.expand_dims(img_array, axis=0)  # Add batch dimension
    return img_array

# Demo prediction function
def predict_image(img_path):
    img_array = load_and_prepare_image(img_path)
    prediction = model.predict(img_array)
    predicted_class = class_names[np.argmax(prediction)]

    # Display image with prediction
    plt.imshow(image.load_img(img_path))
    plt.title(f'Predicted: {predicted_class}')
    plt.axis('off')
    plt.show()

# Run the demo
predict_image('test_images/sample_plastic.jpg')  # Replace with your test image path
